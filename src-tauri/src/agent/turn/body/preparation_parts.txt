                    "transport": diagnostic,

    let setup::PreparedInput {
        request,
        context:
            TurnSkillContext {
                skill_catalog,
                skill_instructions,
                selinux_enabled,
                selinux_scope,

        permission_mode,
        workspace_context,
        global_instructions,

        prompt,

        &cancellation_reason,

    let permission_mode = permission_mode.as_str();

        .as_deref()

        // P7-A：回退 namespace 不再掺 turn_id——掺了它每个 turn 都是新 key，
